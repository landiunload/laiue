add_executable(laiue_animation_demo animation_demo.c)
target_link_libraries(laiue_animation_demo PRIVATE laiue_bootstrap)
target_compile_definitions(laiue_animation_demo PRIVATE LAIUE_STATIC=1)
laiue_configure_precise_fp(laiue_animation_demo)

foreach(_module content model animation)
    if(TARGET laiue_${_module})
        if(LAIUE_MODULE_LIBRARY_TYPE_RESOLVED STREQUAL "STATIC")
            target_link_libraries(laiue_animation_demo PRIVATE laiue_${_module})
            string(TOUPPER "${_module}" _definition)
            target_compile_definitions(laiue_animation_demo PRIVATE
                "LAIUE_ANIMATION_DEMO_STATIC_${_definition}=1")
        else()
            add_dependencies(laiue_animation_demo laiue_${_module})
        endif()
    endif()
endforeach()
if(NOT LAIUE_MODULE_LIBRARY_TYPE_RESOLVED STREQUAL "STATIC")
    target_compile_definitions(laiue_animation_demo PRIVATE LAIUE_ANIMATION_DEMO_DYNAMIC=1)
endif()
if(TARGET laiue_render)
    target_compile_definitions(laiue_animation_demo PRIVATE LAIUE_ANIMATION_DEMO_GRAPHICS=1)
    if(LAIUE_MODULE_LIBRARY_TYPE_RESOLVED STREQUAL "STATIC")
        target_link_libraries(laiue_animation_demo PRIVATE laiue_render)
    else()
        if(LAIUE_RENDER_BACKEND_RESOLVED STREQUAL "D3D12" AND TARGET laiue_graphics_d3d12)
            add_dependencies(laiue_animation_demo laiue_graphics_d3d12)
            target_compile_definitions(laiue_animation_demo PRIVATE LAIUE_ANIMATION_DEMO_D3D12=1)
        elseif(TARGET laiue_graphics_vulkan)
            add_dependencies(laiue_animation_demo laiue_graphics_vulkan)
        endif()
        if(TARGET laiue_graphics_vulkan)
            add_dependencies(laiue_animation_demo laiue_graphics_vulkan)
        endif()
    endif()
endif()
if(TARGET laiue_window)
    target_compile_definitions(laiue_animation_demo PRIVATE LAIUE_ANIMATION_DEMO_WINDOW=1)
    if(LAIUE_MODULE_LIBRARY_TYPE_RESOLVED STREQUAL "STATIC")
        target_link_libraries(laiue_animation_demo PRIVATE laiue_window)
    else()
        add_dependencies(laiue_animation_demo laiue_window)
    endif()
endif()
add_custom_target(laiue_animation_demo_assets
    COMMAND "${CMAKE_COMMAND}" -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets"
        "$<TARGET_FILE_DIR:laiue_animation_demo>/assets_animation"
    VERBATIM)
add_dependencies(laiue_animation_demo laiue_animation_demo_assets)
if(WIN32 AND TARGET laiue_window AND TARGET laiue_render)
    laiue_configure_standalone_executable(laiue_animation_demo AnimationDemoEntryPoint WINDOWED)
else()
    laiue_configure_standalone_executable(laiue_animation_demo AnimationDemoEntryPoint)
endif()
