# Числа произвольной точности. Отдельный модуль, а не часть мира: ими
# пользуются и бесконечные координаты мира, и скорости твёрдых тел в
# physics, а зависеть от мира физика не вправе и не должна.
# PRECISE_FP: модуль точной арифметики не может собираться с /fp:fast.
# Под ним компилятор вправе считать, что NaN и бесконечности не бывает, и
# выбрасывает проверки, на которых держится преобразование из double.
laiue_add_module(numeric
    PRECISE_FP
    RUNTIME_ROLE CORE
    SOURCES
        infinite_coord.c
        infinite_coord.h
        ../api.h
)
